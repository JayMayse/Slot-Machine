else if (choices == "v")
{
    GameCoins -= 3;
    for (int col = 0; col < grid; col++)
    {
        bool colMatch = true;
        for (int row = 1; row < grid; row++)
        {
            if (slots[row, col] != slots[0, col])
            {
                colMatch = false;
                break;
            }
        }
        if (colMatch)
        {
            Console.WriteLine("Congratulations! You won on a vertical line");
            GameCoins += 6;
            isWinner = true;
            break;
        }
    }
}


else if (choices == "h")
{
    GameCoins -= 3;
    for (int row = 0; row < grid; row++)
    {
        bool rowMatch = true;
        for (int col = 1; col < grid; col++)
        {
            if (slots[row, col] != slots[row, 0])
            {
                rowMatch = false;
                break;
            }
        }
        if (rowMatch)
        {
            Console.WriteLine("Congratulations! You won on a horizontal line");
            GameCoins += 6;
            isWinner = true;
            break;
        }
    }
}


else if (choices == "d")
{
    GameCoins -= 5;
    bool leftToRightMatch = true;
    for (int i = 1; i < grid; i++)
    {
        if (slots[i, i] != slots[0, 0])
        {
            leftToRightMatch = false;
            break;
        }
    }

    bool rightToLeftMatch = true;
    for (int i = 1; i < grid; i++)
    {
        if (slots[i, grid - 1 - i] != slots[0, grid - 1])
        {
            rightToLeftMatch = false;
            break;
        }
    }

    if (leftToRightMatch || rightToLeftMatch)
    {
        Console.WriteLine("Congratulations! You won on the diagonal");
        GameCoins += 10;
        isWinner = true;
    }
}
